```javascript
// withTimeout ヘルパー
function withTimeout(promise, ms, onTimeout = null) {
  if (ms == null || ms === Infinity) return promise;
  return Promise.race([
    promise,
    new Promise((_, reject) => setTimeout(() => {
      if (onTimeout) onTimeout();
      reject(new Error(`Timeout after ${ms}ms`));
    }, ms)),
  ]);
}
```

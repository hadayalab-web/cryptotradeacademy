// エンゲージメント率スコア（0-100点）
const engagementScore = Math.min(100, engagementRate * 2000); // 5% = 100点

// インプレッションスコア（0-100点）
const impressionScore = impressions > 0 ? Math.min(100, (impressions / target.max) * 100) : 0;

// フォロワー数スコア（適度な規模を重視、0-50点）
// 10万-50万フォロワーが最適（大きすぎるとエンゲージメント率が下がる）
let followerScore = 0;
if (followerCount >= 10000 && followerCount <= 500000) {
  followerScore = 50;
} else if (followerCount > 500000) {
  followerScore = 30; // 大きすぎるとエンゲージメント率が下がる傾向
} else {
  followerScore = 20; // 小さすぎるとリーチが限定的
}

// 総合スコア（エンゲージメント率を重視）
const totalScore = engagementScore * 0.5 + impressionScore * 0.3 + followerScore * 0.2;
```

**重要ポイント**:

- **エンゲージメント率**: 50%の重み（最重視）
- **インプレッション数**: 30%の重み
- **フォロワー数**: 20%の重み（10万-50万が最適）

function calculateExpectedPerformance(impressions, engagementRate, hasTelegramLink, hasWhopLink) {
  const expectedEngagements = impressions * (engagementRate / 100);

  // Telegramリンクがある場合、クリック率を考慮
  if (hasTelegramLink) {
    // クリック率はエンゲージメント率の10-20%と仮定
    const clickRate = engagementRate * 0.15; // 15%
    expectedEngagements += impressions * (clickRate / 100);
  }

  // Whopリンクがある場合、コンバージョン率を考慮
  if (hasWhopLink) {
    // コンバージョン率はエンゲージメント率の1-5%と仮定
    const conversionRate = engagementRate * 0.03; // 3%
    expectedEngagements += impressions * (conversionRate / 100);
  }

  return {
    expectedEngagements: Math.round(expectedEngagements),
    engagementRate
  };
}
```

### ストックインフルエンサーの期待値

1. **投稿内容の問題**: コンテンツが受動的でインタラクション喚起ゼロ（質問/CTA欠如）

3. **スパムフィルタ接近**: 業界平均比1/100はスパムフィルタ接近
